\section{Academic Components}

\begin{frame}{Mathematical Expression}
    \begin{block}{Regularized Objective}
        \begin{equation*}
            \min_{\theta} \; \mathcal{L}(\theta)
            =
            \frac{1}{n}\sum_{i=1}^{n}
            \left(y_i - f(x_i;\theta)\right)^2
            + \lambda \lVert \theta \rVert_2^2
        \end{equation*}
    \end{block}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \ctextbf{Data fit}
            \begin{itemize}
                \item Average prediction error
                \item Directly tied to observations
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \ctextbf{Complexity control}
            \begin{itemize}
                \item Penalizes unstable parameters
                \item Supports reproducible inference
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
